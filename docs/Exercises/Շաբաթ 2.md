        // Copy Terminal Command
        function copyTerminalCode() {
            const cmdText = document.getElementById('terminalCommand').innerText;
            const copyBtnText = document.getElementById('copyTermText');

            // Copying using fallback / execCommand for iFrame compatibility
            const textarea = document.createElement('textarea');
            textarea.value = cmdText;
            document.body.appendChild(textarea);
            textarea.select();
            document.execCommand('copy');
            document.body.removeChild(textarea);

            copyBtnText.innerText = 'Copied!';
            setTimeout(() => {
                copyBtnText.innerText = 'Copy';
            }, 2000);
        }

        // AI Model Filter
        function filterAI(type) {
            const buttons = document.querySelectorAll('.ai-filter-btn');
            buttons.forEach(btn => btn.classList.remove('active', 'bg-brand-600', 'text-white'));
            
            event.target.classList.add('active', 'bg-brand-600', 'text-white');

            const cards = document.querySelectorAll('.ai-card');
            cards.forEach(card => {
                if (type === 'all') {
                    card.style.display = 'block';
                } else if (type === 'top') {
                    if (card.classList.contains('top-rated')) {
                        card.style.display = 'block';
                    } else {
                        card.style.display = 'none';
                    }
                }
            });
        }
