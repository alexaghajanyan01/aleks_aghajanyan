#### Քայլ 2․ Ֆայլի ստեղծում և ալգորիթմի տեղադրում

VS Code-ում ստեղծվեց `smile_detector.py` ֆայլը հետևյալ տրամաբանությամբ․

1. **Մոդելների ավտոմատ ներբեռնում (`urllib`, `os`)․**
Որպեսզի VS Code-ը կախված չլինի համակարգչի տեղական PATH-երից, ծրագիրը ստուգում է `haarcascade_frontalface_default.xml` և `haarcascade_smile.xml` ֆայլերի առկայությունը թղթապանակում: Եթե չկան, ավտոմատ քաշում է GitHub-ից:
2. **Տեսախցիկի ակտիվացում (`cv2.VideoCapture`)․**
Կոդը միացնում է տեսախցիկը և սկսում կարդալ կանոնավոր կադրեր:
3. **Կադրի մշակում (Grayscale)․**
Գունավոր պատկերը փոխարկվում է սև-սպիտակի (`cv2.COLOR_BGR2GRAY`), որպեսզի պատկերի մշակումը VS Code-ում աշխատի առանց lag-երի:
4. **Դեմքի և Ժպիտի Հայտնաբերում (ROI)․**
Նախ հայտնաբերվում է դեմքը և գծվում է կապույտ ուղղանկյուն: Այնուհետև դեմքի հատվածում (Region of Interest) փնտրվում է ժպիտը:
5. **Կարևոր օպտիմալացում (Week 2 debugging)․**
Ժպիտի ճանաչման զգայունությունը կարգավորվեց `minNeighbors=5` (կամ `20`) պարամետրով, որպեսզի ծրագիրն ավելի հեշտ նկատի ժպիտը և դրա շուրջ գծի կանաչ ուղղանկյուն:

#### Ծրագրի ամբողջական կոդը (`smile_detector.py`)

```python
import cv2
import urllib.request
import os

# XML մոդելների հղումները GitHub-ում
FACE_CASCADE_URL = "https://raw.githubusercontent.com/opencv/opencv/master/data/haarcascades/haarcascade_frontalface_default.xml"
SMILE_CASCADE_URL = "https://raw.githubusercontent.com/opencv/opencv/master/data/haarcascades/haarcascade_smile.xml"

FACE_CASCADE_FILE = "haarcascade_frontalface_default.xml"
SMILE_CASCADE_FILE = "haarcascade_smile.xml"

# Ավտոմատ ստուգում և ներբեռնում
if not os.path.exists(FACE_CASCADE_FILE):
    print("Ներբեռնվում է դեմքի ճանաչման մոդելը...")
    urllib.request.urlretrieve(FACE_CASCADE_URL, FACE_CASCADE_FILE)

if not os.path.exists(SMILE_CASCADE_FILE):
    print("Ներբեռնվում է ժպիտի ճանաչման մոդելը...")
    urllib.request.urlretrieve(SMILE_CASCADE_URL, SMILE_CASCADE_FILE)

# Բեռնում ենք մոդելները
face_cascade = cv2.CascadeClassifier(FACE_CASCADE_FILE)
smile_cascade = cv2.CascadeClassifier(SMILE_CASCADE_FILE)

# Միացնում ենք տեսախցիկը
cap = cv2.VideoCapture(0)

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # Գունավոր պատկերը փոխարկում ենք մոխրագույնի (Grayscale)
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # Հայտնաբերում ենք դեմքը
    faces = face_cascade.detectMultiScale(gray, scaleFactor=1.3, minNeighbors=5)

    for (x, y, w, h) in faces:
        # Գծում ենք կապույտ ուղղանկյուն դեմքի շուրջ
        cv2.rectangle(frame, (x, y), (x + w, y + h), (255, 0, 0), 2)

        # Առանձնացնում ենք դեմքի տիրույթը (ROI - Region of Interest)
        roi_gray = gray[y:y + h, x:x + w]
        roi_color = frame[y:y + h, x:x + w]

        # Փնտրում ենք ժպիտը դեմքի տիրույթում
        smiles = smile_cascade.detectMultiScale(roi_gray, scaleFactor=1.7, minNeighbors=20)

        for (sx, sy, sw, sh) in smiles:
            # Գծում ենք կանաչ ուղղանկյուն ժպիտի շուրջ
            cv2.rectangle(roi_color, (sx, sy), (sx + sw, sy + sh), (0, 255, 0), 2)
            cv2.putText(frame, 'Smiling :)', (x, y - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)

    cv2.imshow('Smile Detector - Real Time', frame)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()

```
